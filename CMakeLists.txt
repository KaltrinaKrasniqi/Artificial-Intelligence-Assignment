cmake_minimum_required(VERSION 3.16)
project(ProjektAI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(latin_square problems/latin_square/main.cpp problems/latin_square/solver.cpp problems/latin_square/tests.cpp problems/latin_square/validator.cpp)
target_include_directories(latin_square PRIVATE shared)
if(MSVC)
  target_compile_options(latin_square PRIVATE /W4)
else()
  target_compile_options(latin_square PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME latin_square_tests COMMAND latin_square --test)
